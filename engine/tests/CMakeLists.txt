cmake_minimum_required(VERSION 3.16)
project(nexterm-ssh-tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBSSH2 REQUIRED IMPORTED_TARGET libssh2)
enable_testing()
add_executable(ssh_auth_test ssh_auth_test.c)
target_link_libraries(ssh_auth_test PRIVATE PkgConfig::LIBSSH2)
target_compile_options(ssh_auth_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME ssh_auth_unit COMMAND ssh_auth_test)
add_executable(ssh_auth_client ssh_auth_client.c ../src/net/ssh_auth.c)
target_link_libraries(ssh_auth_client PRIVATE PkgConfig::LIBSSH2)
target_compile_options(ssh_auth_client PRIVATE -Wall -Wextra -Werror)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME ssh_auth_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/ssh_auth_integration.py $<TARGET_FILE:ssh_auth_client>)
